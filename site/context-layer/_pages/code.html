quickstart

run one complete exchange.

import {
  decideContextRequest,
  issueScopedBundle,
  writeReceipt,
} from './context-layer-reference.mjs';

const decision = decideContextRequest(request, policy);
const bundle = issueScopedBundle({
  request,
  decision,
  claims,
  issuer: 'did:example:alice',
});
const receipt = writeReceipt({
  operation: 'bundle.issue',
  request,
  decision,
  bundle,
});

the reference runtime accepts a request & policy, reduces the requested selectors, issues an expiring bundle, then writes a payload-minimized receipt.
